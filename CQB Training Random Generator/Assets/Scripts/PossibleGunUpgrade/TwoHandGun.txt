using UdonSharp;
using UnityEngine;
using VRC.SDKBase;

[UdonBehaviourSyncMode(BehaviourSyncMode.Continuous)]
public class TwoHandGun : UdonSharpBehaviour
{
    [Header("Gun References")]
    public VRC_Pickup pistol;          // the pickup on this same object
    public VRC_Pickup foregrip;        // ForegripHandle's pickup
    public Transform gunInner;
    public Transform foregripAnchor;
    public ParticleSystem particle;
    public AudioSource audioSource;
    public AudioClip gunshotClip;
    [Header("Fire Settings")]
    public bool fullAuto = false;
    public int fireRate = 600; // rounds per minute

    private float nextFireTime;
    private bool triggerWasHeld;


    [UdonSynced] private bool syncTwoHand;
    [UdonSynced] private Vector3 syncAimLocal;   // aim direction in GunRoot local space

    private Vector3 restDirLocal;
    private Quaternion restLook;

    void Start()
    {
        // assumes gunInner is at identity local rotation in the prefab
        restDirLocal = transform.InverseTransformPoint(foregripAnchor.position);
        restLook = Quaternion.LookRotation(restDirLocal, Vector3.up);
        foregrip.pickupable = false;
    }

    public override void OnPickup()   // pistol picked up (local player is holder)
    {
        foregrip.pickupable = true;
    }

    public override void OnDrop()
    {
        if (foregrip.IsHeld) foregrip.Drop();
        foregrip.pickupable = false;
        syncTwoHand = false;
    }

    public override void PostLateUpdate()
    {
        bool local = Networking.IsOwner(gameObject) && pistol.IsHeld;
        Vector3 aimLocal = restDirLocal;
        bool two = false;

        if (local)
        {
            two = foregrip.IsHeld;
            if (two)
            {
                Vector3 dWorld = foregrip.transform.position - transform.position;
                aimLocal = transform.InverseTransformDirection(dWorld);
            }
            syncTwoHand = two;
            syncAimLocal = aimLocal;
        }
        else
        {
            two = syncTwoHand;
            if (two) aimLocal = syncAimLocal;
        }

        // aim
        Quaternion target = Quaternion.identity;
        if (two && aimLocal.sqrMagnitude > 0.0001f
            && Mathf.Abs(Vector3.Dot(aimLocal.normalized, Vector3.up)) < 0.999f)
        {
            target = Quaternion.LookRotation(aimLocal, Vector3.up) * Quaternion.Inverse(restLook);
        }
        gunInner.localRotation = local
            ? target
            : Quaternion.Slerp(gunInner.localRotation, target, 0.3f); // smooth remote jitter

        // park the handle at the anchor when not held
        if (!foregrip.IsHeld)
        {
            foregrip.transform.SetPositionAndRotation(
                foregripAnchor.position, foregripAnchor.rotation);
        }
    }

    private void Update()
    {
        if (!fullAuto || !triggerWasHeld)
            return;

        if (pistol == null || !pistol.IsHeld || pistol.currentPlayer != Networking.LocalPlayer)
        {
            triggerWasHeld = false;
            return;
        }

        if (!Networking.IsOwner(gameObject))
        {
            triggerWasHeld = false;
            return;
        }

        TryFire();
    }

    public override void OnPickupUseDown()
    {
        if (pistol == null || !pistol.IsHeld || pistol.currentPlayer != Networking.LocalPlayer)
            return;

        if (!Networking.IsOwner(gameObject))
            return;

        triggerWasHeld = true;
        TryFire();
    }

    public override void OnPickupUseUp()
    {
        triggerWasHeld = false;
    }

    private void TryFire()
    {
        if (Time.time < nextFireTime)
            return;

        nextFireTime = Time.time + 60f / fireRate;

        SendCustomNetworkEvent(VRC.Udon.Common.Interfaces.NetworkEventTarget.All, nameof(Fire));
    }

    public void Fire()
    {
        if (particle == null)
            return;

        particle.Emit(1);

        if (audioSource != null && gunshotClip != null)
        {
            audioSource.PlayOneShot(gunshotClip);
        }
    }
}
